export async function deleteRecording(
  scope: string,
  id: string,
  expected: SavedRecording,
): Promise<void> {
  const db = await open();
  return new Promise((resolve, reject) => {
    const tx = db.transaction("pending", "readwrite");
    const s = tx.objectStore("pending");
    const r = s.get(key(scope, id, expected));
    r.onsuccess = () => {
      const current = r.result as SavedRecording | undefined;
      if (
        current?.version === expected.version &&
        current?.responseId === expected.responseId
      )
        s.delete(key(scope, id, expected));
    };
    tx.oncomplete = () => {
      db.close();
      resolve();
    };
    tx.onerror = tx.onabort = () => {
      db.close();
      reject(tx.error);
    };
  });
}
export function selectRecordingQuestion(
  state: any,
  recordingQuestion: any,
  pendingIds: string[],
  savedShort: (id: string) => unknown,
) {
  return (
    recordingQuestion ||
    state?.questions.find(
      (q: any) =>
        (pendingIds.includes(q.id) ||
          (q.max_seconds !== 120 && savedShort(q.id))) &&
        !state.responses.some(
          (r: any) => r.question_id === q.id && r.status !== "recording",
        ),
    ) ||
    (state?.class.status === "active"
      ? state.questions.find((q: any) => q.status === "active")
      : null)
  );
}
